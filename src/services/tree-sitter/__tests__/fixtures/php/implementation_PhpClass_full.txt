/**
 * PhpClass is a sample class.
 */
class PhpClass
{
    private string $name;

    public function __construct(string $name)
    {
        $this->name = $name;
    }

    /**
     * getName returns the name.
     */
    public function getName(): string
    {
        return $this->name;
    }

    public function calculate(int $a, int $b): int
    {
        return $a + $b;
    }
}
